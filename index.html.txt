        async function fetchGeminiWithRetry(prompt, systemInstruction, retries = 5) {
            const endpoint = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`;
            const payload = {
                contents: [{ parts: [{ text: prompt }] }],
                systemInstruction: { parts: [{ text: systemInstruction }] }
            };

            const delays = [1000, 2000, 4000, 8000, 16000];

            for (let i = 0; i < retries; i++) {
                try {
                    const response = await fetch(endpoint, {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify(payload)
                    });

                    if (!response.ok) throw new Error(`HTTP erro! status: ${response.status}`);

                    const data = await response.json();
                    return data.candidates?.[0]?.content?.parts?.[0]?.text || "Desculpe, não consegui gerar uma resposta.";
                } catch (error) {
                    if (i === retries - 1) return "Erro de conexão ao contatar a inteligência artificial. Por favor, tente novamente mais tarde.";
                    await new Promise(resolve => setTimeout(resolve, delays[i]));
                }
            }
        }

        function renderMCQuestions() {
            const container = document.getElementById('mc-container');
            container.innerHTML = '';
            mcQuestions.forEach((q, index) => {
                const qDiv = document.createElement('div');
                qDiv.className = 'bg-white p-6 rounded-lg shadow-sm border border-gray-200';
                qDiv.id = `mc-box-${q.id}`;
                
                let html = `
                    <div class="flex justify-between items-start mb-4 border-b pb-2">
                        <span class="font-bold text-gray-500">Questão ${index + 1}</span>
                        <span class="text-xs bg-gray-100 text-gray-600 px-2 py-1 rounded border">${q.category}</span>
                    </div>
                    <p class="text-gray-800 mb-6 font-medium">${q.text}</p>
                    <div class="space-y-3" id="options-${q.id}">
                `;

                for (const [key, value] of Object.entries(q.options)) {
                    html += `
                        <button onclick="handleMCAnswer('${q.id}', '${key}')" id="btn-${q.id}-${key}" class="option-btn w-full text-left p-4 border rounded-md text-sm text-gray-700 bg-gray-50">

                `;
                qDiv.innerHTML = html;

        function handleMCAnswer(qId, selectedKey) {
            if (state.mcAnswers[qId]) return; // Already answered
            
            const q = mcQuestions.find(x => x.id === qId);
            const isCorrect = selectedKey === q.correct;
            
            state.mcAnswers[qId] = { selected: selectedKey, isCorrect: isCorrect };
            updateStats(q.category, isCorrect);
            saveState();

            // UI Updates
            const buttons = document.querySelectorAll(`#options-${qId} button`);
            buttons.forEach(btn => {
                btn.disabled = true;
                btn.classList.add('cursor-not-allowed', 'opacity-60');
            });

            const selectedBtn = document.getElementById(`btn-${qId}-${selectedKey}`);
            const correctBtn = document.getElementById(`btn-${qId}-${q.correct}`);
            
            selectedBtn.classList.remove('bg-gray-50', 'opacity-60');
            correctBtn.classList.remove('bg-gray-50', 'opacity-60');
            
            if (isCorrect) {
                selectedBtn.classList.add('bg-green-100', 'border-green-500', 'text-green-800');
            } else {
                selectedBtn.classList.add('bg-red-100', 'border-red-500', 'text-red-800');
                correctBtn.classList.add('bg-green-100', 'border-green-500', 'text-green-800'); // Show correct
            }

            // Gabarito Comentado Focus no Erro/Acerto
            const feedbackDiv = document.getElementById(`feedback-${qId}`);
            feedbackDiv.classList.remove('hidden');
            
            if (isCorrect) {
                feedbackDiv.classList.add('bg-green-50', 'border-green-200', 'text-green-800');
                feedbackDiv.innerHTML = `<strong>Acertou!</strong> ${q.explanations[selectedKey]}`;
            } else {
                feedbackDiv.classList.add('bg-red-50', 'border-red-200', 'text-red-800');
                feedbackDiv.innerHTML = `<strong>Errou. Entenda a pegadinha:</strong><br>Você marcou ${selectedKey}: ${q.explanations[selectedKey]}<br><br><strong>A correta é ${q.correct}:</strong> ${q.explanations[q.correct]}`;
            }
        }

        // --- STATS & CHART ---
        function updateStats(categoryRaw, isCorrect) {
            state.stats.total++;
            if (!isCorrect) return; // Only count correct for the radar

            // Categorize broadly
            const cat = categoryRaw.toLowerCase();
            if (cat.includes('hist') || cat.includes('cnv')) state.stats.history++;
            else if (cat.includes('interp') || cat.includes('lit')) state.stats.interp++;
            else state.stats.grammar++;
        }

        function updateChart() {
            const ctx = document.getElementById('performanceChart');
            
            const totalAnswered = Object.keys(state.mcAnswers).length + Object.keys(state.ceAnswers).length;
            const textDiv = document.getElementById('stats-text');
            
            if (totalAnswered === 0) {
                textDiv.innerHTML = "Responda algumas questões para ver seu gráfico!";
                return;
            }

            const totalCorrect = state.stats.history + state.stats.interp + state.stats.grammar;
            const percent = Math.round((totalCorrect / totalAnswered) * 100);
            
            textDiv.innerHTML = `
                <div class="text-2xl font-bold text-gray-800 mb-1">Precisão: ${percent}%</div>
                <div class="text-sm">Respondidas: ${totalAnswered} / ${state.stats.max}</div>
                <div class="text-sm mt-2 text-gray-500">Continue usando o gabarito comentado para aprender com os erros!</div>
            `;

            if (myChart) myChart.destroy();

            // Calculate max possible per category (approximate for radar max)
            const data = [state.stats.interp, state.stats.history, state.stats.grammar];

            myChart = new Chart(ctx, {
                type: 'radar',
                data: {
                    labels: ['Interpretação/Literatura', 'História/Contexto', 'Gramática/Sintaxe'],
                    datasets: [{
                        label: 'Acertos por Área',
                        data: data,
                        backgroundColor: 'rgba(217, 119, 6, 0.2)', // Amber-600
                        borderColor: 'rgba(217, 119, 6, 1)',
                        pointBackgroundColor: 'rgba(217, 119, 6, 1)',
                        pointBorderColor: '#fff',
                        pointHoverBackgroundColor: '#fff',
                        pointHoverBorderColor: 'rgba(217, 119, 6, 1)',
                        borderWidth: 2
                    }]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false, // Critical for Tailwind container constraints
                    scales: {
                        r: {
                            angleLines: { color: 'rgba(0,0,0,0.1)' },
                            grid: { color: 'rgba(0,0,0,0.1)' },
                            pointLabels: {
                                font: { size: 12, family: "'Open Sans', sans-serif" },
                                color: '#4B5563'
                            },
                            ticks: {
                                beginAtZero: true,
                                stepSize: 2,
                                display: false // hide numbers on rings
                            }
                        }
                    },
                    plugins: {
                        legend: { display: false },
                        tooltip: {
                            callbacks: {
                                label: function(context) {
                                    return ` ${context.raw} Acertos`;
                                }
                            }

                    }
                }
            });

                const q = mcQuestions.find(x => x.id === qId);
                if(q) {
                    const buttons = document.querySelectorAll(`#options-${qId} button`);
                    buttons.forEach(btn => {
                        btn.disabled = true;
                        btn.classList.add('cursor-not-allowed', 'opacity-60');
                    });
                    const selectedBtn = document.getElementById(`btn-${qId}-${ans.selected}`);
                    const correctBtn = document.getElementById(`btn-${qId}-${q.correct}`);
                    if(selectedBtn) selectedBtn.classList.remove('bg-gray-50', 'opacity-60');
                    if(correctBtn) correctBtn.classList.remove('bg-gray-50', 'opacity-60');
                    
                    if (ans.isCorrect) {
                        if(selectedBtn) selectedBtn.classList.add('bg-green-100', 'border-green-500', 'text-green-800');
                    } else {
                        if(selectedBtn) selectedBtn.classList.add('bg-red-100', 'border-red-500', 'text-red-800');
                        if(correctBtn) correctBtn.classList.add('bg-green-100', 'border-green-500', 'text-green-800');
                    }
                    const feedbackDiv = document.getElementById(`feedback-${qId}`);
                    feedbackDiv.classList.remove('hidden');
                    if (ans.isCorrect) {
                        feedbackDiv.classList.add('bg-green-50', 'border-green-200', 'text-green-800');
                        feedbackDiv.innerHTML = `<strong>Acertou!</strong> ${q.explanations[ans.selected]}`;
                    } else {
                        feedbackDiv.classList.add('bg-red-50', 'border-red-200', 'text-red-800');
                        feedbackDiv.innerHTML = `<strong>Errou. Entenda a pegadinha:</strong><br>Você marcou ${ans.selected}: ${q.explanations[ans.selected]}<br><br><strong>A correta é ${q.correct}:</strong> ${q.explanations[q.correct]}`;
                    }
                }
            });

            // Restore CE UI
            Object.keys(state.ceAnswers).forEach(qId => {
                const ans = state.ceAnswers[qId];
                const q = ceQuestions.find(x => x.id === qId);
                if(q) {
